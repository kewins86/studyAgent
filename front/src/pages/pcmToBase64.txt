function pcmToBase64(pcmData) {
		const bytes = new Uint8Array(pcmData.buffer)
		let binary = ''
		for (let i = 0; i < bytes.byteLength; i++) {
		binary += String.fromCharCode(bytes[i])
	}
		return btoa(binary)
}
