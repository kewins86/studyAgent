function stopStepRecording(msgIndex, stepIndex) {
		const msg = messages.value[msgIndex]
		const step = msg.steps[stepIndex]
		step.recording = false

		// #ifdef H5
		stopH5RecordingForStep(msgIndex, stepIndex)
		// #endif

		// #ifndef H5
		if (recorderManager) {
			recorderManager.stop()
		}
		// #endif
}
